cmake_minimum_required(VERSION 3.25)
project(au13xx_startup_probe LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This development probe requires Linux")
endif()
set(CERFENGINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../libCerfEngine" engine)

set(probe_sources
    probe_host.cpp probe_system.cpp probe_gpic.cpp probe_bus.cpp
    probe_dma.cpp probe_dma_cursor.cpp
    probe_uart.cpp probe_sys.cpp probe_nand.cpp probe_trace.cpp host_glue.cpp)
function(probe_target name)
    target_compile_features(${name} PRIVATE cxx_std_20)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    target_include_directories(${name} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/../../libCerfEngine/tests")
    target_link_libraries(${name} PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,cerfengine::cerfengine>")
endfunction()
add_executable(au13xx_startup_probe main.cpp ${probe_sources})
probe_target(au13xx_startup_probe)
enable_testing()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_test(NAME synthetic_nand_paths
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/synthetic_nand_test.py")
add_executable(au13xx_probe_test probe_test.cpp ${probe_sources})
probe_target(au13xx_probe_test)
add_test(NAME au13xx_probe_test COMMAND au13xx_probe_test)
add_executable(au13xx_dma_test dma_test.cpp ${probe_sources})
probe_target(au13xx_dma_test)
add_test(NAME au13xx_dma_test COMMAND au13xx_dma_test)
add_test(NAME reserved_2d_decrement COMMAND au13xx_dma_test --reserved-2d-decrement)
add_test(NAME reserved_2d_byte COMMAND au13xx_dma_test --reserved-2d-byte)
add_test(NAME reserved_2d_halfword COMMAND au13xx_dma_test --reserved-2d-halfword)
add_test(NAME unsupported_burstable_fifo COMMAND au13xx_dma_test --unsupported-burstable-fifo)
add_test(NAME unsupported_gpio_dma COMMAND au13xx_dma_test --unsupported-gpio-dma)
set_tests_properties(reserved_2d_decrement reserved_2d_byte reserved_2d_halfword
    unsupported_burstable_fifo unsupported_gpio_dma PROPERTIES WILL_FAIL TRUE)
add_executable(au13xx_uart_test uart_test.cpp ${probe_sources})
probe_target(au13xx_uart_test)
add_test(NAME au13xx_uart_test COMMAND au13xx_uart_test)
add_test(NAME unsupported_uart_bidir_gate COMMAND au13xx_uart_test --bidir-gate)
add_test(NAME unsupported_uart_open_drain COMMAND au13xx_uart_test --open-drain)
set_tests_properties(unsupported_uart_bidir_gate unsupported_uart_open_drain PROPERTIES WILL_FAIL TRUE)
add_executable(au13xx_execution_test execution_test.cpp ${probe_sources})
probe_target(au13xx_execution_test)
add_test(NAME au13xx_execution_test COMMAND au13xx_execution_test)
add_test(NAME unsupported_armed_watch COMMAND au13xx_execution_test --armed-watch)
set_tests_properties(unsupported_armed_watch PROPERTIES WILL_FAIL TRUE)
add_executable(au13xx_clock_test clock_test.cpp ${probe_sources})
probe_target(au13xx_clock_test)
add_test(NAME au13xx_clock_test COMMAND au13xx_clock_test)
add_test(NAME unsupported_external_counter COMMAND au13xx_clock_test --external-counter)
set_tests_properties(unsupported_external_counter PROPERTIES WILL_FAIL TRUE)
add_executable(au13xx_system_test system_test.cpp ${probe_sources})
probe_target(au13xx_system_test)
add_test(NAME au13xx_system_test COMMAND au13xx_system_test)
add_test(NAME unsupported_high_physical COMMAND au13xx_system_test --high-physical)
add_test(NAME unsupported_reverse_endian COMMAND au13xx_system_test --reverse-endian)
add_test(NAME unsupported_deferred_watch COMMAND au13xx_system_test --deferred-watch)
add_test(NAME unsupported_instruction_watch COMMAND au13xx_system_test --instruction-watch)
set_tests_properties(unsupported_high_physical unsupported_reverse_endian
    unsupported_deferred_watch unsupported_instruction_watch PROPERTIES WILL_FAIL TRUE)
add_test(NAME nand_sequential_block_boundary COMMAND au13xx_probe_test --sequential-cross-block)
add_test(NAME nand_random_block_boundary COMMAND au13xx_probe_test --random-cross-block)
set_tests_properties(nand_sequential_block_boundary nand_random_block_boundary PROPERTIES WILL_FAIL TRUE)
